
# Problem 1: String alphabet analysis

import string

# (1) Introduce a string of exactly 300 characters
text = """In 2025, a group of 12 researchers travelled to an ancient village.
They discovered 7 old buildings, 34 clay pots and 19 stone tools.
The team spent 6 days examining the site and recording every discovery.
Their work helped explain how people lived, traded and worked together.
The researchers also found maps, coins and letters hidden beneath the ground.
After 3 weeks, they returned home with 25 boxes of carefully collected objects.
The discovery attracted attention from historians around the world."""

# Adjust the string to exactly 300 characters if necessary
text = text[:300]

print("Original text:")
print(text)
print("=" * 60)

print("1.1: Number of characters:", len(text))
print("1.1: Number of words:", len(text.split()))

# (2) Reverse, replace b with X, and convert to lowercase
changed_text = text[::-1].replace("b", "X").lower()

print("\n1.2: Reversed and modified text:")
print(changed_text)

# (3) Count the letter a
print("\n1.3: Letter frequency")
print("The letter 'a' appears {} times.".format(text.lower().count("a")))

alphabet = string.ascii_lowercase
missing_letters = []

for letter in alphabet:
    frequency = text.lower().count(letter)
    print("The letter '{}' appears {} times.".format(letter, frequency))

    if frequency == 0:
        missing_letters.append(letter)

print("\nLetters missing from the text:")
print(missing_letters)

# (4) Remove digits 0 through 8 and replace 9 with nine
new_text = ""

for character in text:
    if character == "9":
        new_text += "nine"
    elif character not in "012345678":
        new_text += character

print("\n1.4: Text after changing the digits:")
print(new_text)
